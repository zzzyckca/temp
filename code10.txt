@echo off
cd /d "%~dp0"

echo [1/3] Reading config.ini...
for /f "usebackq eol=; tokens=1,* delims==" %%a in ("config.ini") do (
    if not "%%b"=="" set "%%a=%%b"
)

echo [2/3] Checking Directory: %PREFECT_HOME%
:: 如果資料夾唔存在，就整返個，仲要寫入一個測試檔睇吓有無 Permission
if not exist "%PREFECT_HOME%" mkdir "%PREFECT_HOME%"
echo test > "%PREFECT_HOME%\perm_test.txt"
if exist "%PREFECT_HOME%\perm_test.txt" (
    del "%PREFECT_HOME%\perm_test.txt"
    echo [OK] Permission verified.
) else (
    echo [ERROR] No write permission to %PREFECT_HOME%!
    pause
    exit
)

echo [3/3] Starting Services (Minimizing to Taskbar)...

:: 先開 Server
start "Prefect_Server" /min cmd /c "call "%VENV_DIR%\Scripts\activate.bat" && set PREFECT_HOME=%PREFECT_HOME% && prefect server start"

timeout /t 8 /nobreak >nul

:: 再開 Worker
start "Prefect_Worker" /min cmd /c "call "%VENV_DIR%\Scripts\activate.bat" && set PREFECT_HOME=%PREFECT_HOME% && prefect worker start --pool %WORK_POOL_NAME%"

echo.
echo [Finish] Services launched. 
echo If the windows in taskbar remain open, it's working!
echo UI URL: http://127.0.0.1:4200
pause
